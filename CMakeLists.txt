cmake_minimum_required(VERSION 3.25)
project(EndfieldDisconnectMeme VERSION 0.3.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/permissive->)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Private static MinHook" FORCE)
add_subdirectory(third_party/minhook EXCLUDE_FROM_ALL)
add_library(DisconnectMeme SHARED src/plugin.cpp src/managed.cpp)
target_include_directories(DisconnectMeme PRIVATE src sdk/include)
target_link_libraries(DisconnectMeme PRIVATE minhook)
set_target_properties(DisconnectMeme PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package/$<CONFIG>/disconnect-meme")
add_custom_target(package ALL DEPENDS DisconnectMeme)
foreach(f mod.ini config.ini icon.png zml-package.json)
  add_custom_command(TARGET package POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/mod/${f}" "${CMAKE_BINARY_DIR}/package/$<CONFIG>/disconnect-meme/${f}")
endforeach()
add_custom_command(TARGET package POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/third_party/minhook/LICENSE.txt" "${CMAKE_BINARY_DIR}/package/$<CONFIG>/disconnect-meme/MINHOOK-LICENSE.txt")
enable_testing()
add_test(NAME icon COMMAND powershell -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tools/test-icon.ps1")
add_executable(StateTests tests/state_tests.cpp)
target_include_directories(StateTests PRIVATE src)
add_test(NAME state COMMAND StateTests)
add_library(GameAssembly SHARED tests/managed_fixture.cpp)
target_compile_options(GameAssembly PRIVATE /Ob0)
target_link_options(GameAssembly PRIVATE /OPT:NOICF)
set(ZML_KEYBINDS_DLL "" CACHE FILEPATH "External public Keybinds.dll for black-box integration")
if(ZML_KEYBINDS_DLL)
if(NOT EXISTS "${ZML_KEYBINDS_DLL}")
  message(FATAL_ERROR "Build the independent Keybinds mod, pass -DZML_KEYBINDS_DLL=.../Keybinds.dll")
endif()
add_executable(HookTests tests/hook_tests.cpp)
add_library(ServiceFixture SHARED tests/service_fixture.cpp)
target_include_directories(ServiceFixture PRIVATE sdk/include)
set_target_properties(ServiceFixture PROPERTIES OUTPUT_NAME ZMLRuntime RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/fixture/$<CONFIG>")
target_compile_definitions(HookTests PRIVATE ZML_KEYBINDS_DLL="${ZML_KEYBINDS_DLL}")
target_include_directories(HookTests PRIVATE sdk/include)
target_link_libraries(HookTests PRIVATE minhook)
add_test(NAME native_hooks COMMAND HookTests "$<TARGET_FILE:GameAssembly>" "$<TARGET_FILE:DisconnectMeme>" "$<TARGET_FILE:ServiceFixture>")
foreach(mode dependency signature fastsignature fastfieldreadonly field sessionfield entry ambiguous rollback)
  add_test(NAME native_reject_${mode} COMMAND HookTests "$<TARGET_FILE:GameAssembly>" "$<TARGET_FILE:DisconnectMeme>" "$<TARGET_FILE:ServiceFixture>" ${mode})
endforeach()
set_tests_properties(native_hooks PROPERTIES TIMEOUT 30)
endif()
set(ZML_SERVICES_TEST_EXE "" CACHE FILEPATH "Optional external public services fixture")
set(ZML_LUPA_PATH "" CACHE PATH "Optional already installed Lupa")
if(ZML_SERVICES_TEST_EXE AND ZML_LUPA_PATH)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME public_services COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/services_test.py" "${ZML_SERVICES_TEST_EXE}" "${ZML_LUPA_PATH}")
endif()
